cmake_minimum_required(VERSION 3.16)
# Also supports a standalone editor build without SDL dependencies.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ImperaWorkshop LANGUAGES C CXX)
    include(CTest)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/../src/vars.c" "${CMAKE_CURRENT_SOURCE_DIR}/../src/6000.c")
# Reuse the engine's equipment names rather than maintaining another authoring table.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../src/vars.c" engine_names_source)
string(FIND "${engine_names_source}" "char* D_17f6" equipment_begin)
string(FIND "${engine_names_source}" "char* D_1856" equipment_end)
math(EXPR equipment_length "${equipment_end}-${equipment_begin}")
string(SUBSTRING "${engine_names_source}" ${equipment_begin} ${equipment_length} equipment_source)
string(REGEX MATCHALL "_STATIC_TEXT\\([^,]+, \"[^\"]*\"\\)" equipment_matches "${equipment_source}")
list(LENGTH equipment_matches equipment_count)
if(NOT equipment_count EQUAL 48)
    message(FATAL_ERROR "Expected the engine's 48 equipment names")
endif()
set(equipment_header "static const char *engineEquipmentNames[] = {\n")
foreach(item IN LISTS equipment_matches)
    string(REGEX REPLACE ".*, (\"[^\"]*\").*" "\\1" item_name "${item}")
    string(APPEND equipment_header "${item_name},\n")
endforeach()
string(APPEND equipment_header "};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/inventory_names.h" "${equipment_header}")
add_library(impera_workshop_core STATIC resource_document.cpp text_preview.cpp formats.cpp dialogue.cpp map_editor.cpp project.cpp ../src/mod/package.c)
target_include_directories(impera_workshop_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(impera_workshop_core PUBLIC Qt6::Widgets)
target_include_directories(impera_workshop_core PUBLIC . ../src)
add_executable(ImperaWorkshop WIN32 MACOSX_BUNDLE main.cpp window.cpp dialogue_editor.cpp)
target_link_libraries(ImperaWorkshop PRIVATE impera_workshop_core)
set_target_properties(ImperaWorkshop PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Impera Workshop"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.impera.workshop" MACOSX_BUNDLE_ICON_FILE impera.icns)
if(WIN32)
    enable_language(RC)
    target_sources(ImperaWorkshop PRIVATE ../packaging/impera.rc)
    target_include_directories(ImperaWorkshop PRIVATE ../packaging)
elseif(APPLE)
    target_sources(ImperaWorkshop PRIVATE ../packaging/impera.icns)
    set_source_files_properties(../packaging/impera.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
endif()
# Embed the canonical generated icon; no runtime artwork path dependency.
qt_add_resources(ImperaWorkshop workshop_icon PREFIX "/" BASE ../packaging FILES ../packaging/impera.png)
install(TARGETS ImperaWorkshop BUNDLE DESTINATION . RUNTIME DESTINATION bin)
option(IMPERA_WORKSHOP_DEPLOY_QT "Install portable Workshop Qt runtime dependencies" OFF)
if(IMPERA_WORKSHOP_DEPLOY_QT)
    if(Qt6_VERSION VERSION_LESS 6.7)
        message(FATAL_ERROR "Portable Workshop deployment requires Qt 6.7 or newer")
    endif()
    qt_generate_deploy_app_script(TARGET ImperaWorkshop OUTPUT_SCRIPT workshop_deploy_script
        NO_TRANSLATIONS NO_COMPILER_RUNTIME)
    if(MSVC)
        set(CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS_SKIP TRUE)
        include(InstallRequiredSystemLibraries)
        install(FILES ${CMAKE_INSTALL_SYSTEM_RUNTIME_LIBS} DESTINATION bin)
    endif()
    install(SCRIPT ${workshop_deploy_script})
endif()
if(BUILD_TESTING OR U5D_BUILD_TESTS)
    enable_testing()
    # Compile the actual engine keyword matcher in isolation as a compatibility oracle.
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../src/6000.c" matcher_source)
    string(FIND "${matcher_source}" "int ULTIMA_6f1e(" matcher_begin)
    string(FIND "${matcher_source}" "// ASM\nstatic int ULTIMA_6f90" matcher_end)
    math(EXPR matcher_length "${matcher_end}-${matcher_begin}")
    string(SUBSTRING "${matcher_source}" ${matcher_begin} ${matcher_length} matcher_function)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/engine_match_oracle.c"
        "#include <string.h>\ntypedef unsigned char byte;\nstatic byte DAT_000738d8[128];\n${matcher_function}\nint WorkshopEngineMatch(const byte*key,char*input){for(int i='a';i<='z';++i)DAT_000738d8[i]=2;return ULTIMA_6f1e((byte*)key,input); }\n")
    add_executable(impera_workshop_test "${CMAKE_CURRENT_BINARY_DIR}/engine_match_oracle.c" tests.cpp window.cpp dialogue_editor.cpp engine_codec_test.c ../src/common/lzw.c)
    target_link_libraries(impera_workshop_test PRIVATE impera_workshop_core)
    add_test(NAME workshop COMMAND impera_workshop_test)
    set_tests_properties(workshop PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
endif()

if(MSVC)
    foreach(target impera_workshop_core ImperaWorkshop impera_workshop_test)
        if(TARGET ${target})
            target_compile_options(${target} PRIVATE /utf-8)
            target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
        endif()
    endforeach()
endif()
